\begin{tikzpicture}[

  dec/.style={draw=sadRed, line width=1pt, diamond, aspect=2.4, fill=sadRed!8,

  bad/.style={draw=sadRed, line width=1pt, rounded corners, fill=sadRed!12,

\node[box] (in) {Input from \texttt{computeCov2D} (dilated)\\[3pt]
  $\Sigma''=\begin{pmatrix}\Sigma''_{00} & \Sigma'_{01}\\ \Sigma'_{01} & \Sigma''_{11}\end{pmatrix}$};

\node[dec, below=1.1cm of in] (det) {$\det=\Sigma''_{00}\,\Sigma''_{11}-{\Sigma'_{01}}^{2}$\\[2pt]$\det = 0\ ?$};

\node[bad, below left=1.0cm and 1.2cm of det] (cull) {Gaussian is culled\\(not rendered)};

\node[good, below=1.1cm of det] (conic) {$\Sigma'^{-1}=\mathrm{conic}=\dfrac{1}{\det}\begin{pmatrix}\Sigma''_{11} & -\Sigma'_{01}\\ -\Sigma'_{01} & \Sigma''_{00}\end{pmatrix}$\\[4pt]
  stored as $(\mathrm{conic}.x,\ \mathrm{conic}.y,\ \mathrm{conic}.z)=\left(\dfrac{\Sigma''_{11}}{\det},\ \dfrac{-\Sigma'_{01}}{\det},\ \dfrac{\Sigma''_{00}}{\det}\right)$};

\draw[arr] (in) -- (det);
\draw[arr] (det.south) -- node[right, font=\small, text=greyT] {no} (conic.north);
\draw[arr] (det.west) -- node[above, font=\small, text=greyT] {yes} (cull.east);

\end{tikzpicture}
